02

Architektura — N-Tier / Clean Architecture

Projekt stosuje architekturę warstwową z czytelnym podziałem odpowiedzialności. Żadna warstwa nie może znać szczegółów warstwy powyżej niej.

KuchniaUCygana.Web  ·  ASP.NET MVC
Controllers · Views · ViewModels · wwwroot
↓   używa DTOs z Application
KuchniaUCygana.Application  ·  BLL
Services · DTOs · Interfaces · AutoMapper Profiles · Validators (FluentValidation)
↓   używa Interfejsów z Domain
KuchniaUCygana.Domain  ·  Core
Entities · Repository Interfaces · Domain Events
↓   implementuje Interfejsy
KuchniaUCygana.Infrastructure  ·  DAL + Zewnętrzne
Repositories · DB (Dapper) · Migrations · External APIs: Stripe, Maps, OpenAI · File Storage · PDF Generator · Cache
ℹ️
Zasada: Warstwa Web zna tylko Application. Application zna tylko Domain. Infrastructure implementuje interfejsy Domain — nigdy na odwrót.

5.2 Generic Repository

csharp 
// Domain/Interfaces/IRepository.cs
public interface IRepository<T> where T : BaseEntity
{
    Task<T?> GetByIdAsync(int id);
    Task<IEnumerable<T>> GetAllAsync();
    Task<int> InsertAsync(T entity);
    Task<bool> UpdateAsync(T entity);
    Task<bool> DeleteAsync(int id);
}

// Infrastructure/Persistence/Repositories/BaseRepository.cs
public abstract class BaseRepository<T> : IRepository<T> where T : BaseEntity
{
    protected readonly IDbConnectionFactory _factory;

    protected BaseRepository(IDbConnectionFactory factory) => _factory = factory;

    public async Task<T?> GetByIdAsync(int id)
    {
        using var db = _factory.CreateConnection();
        return await db.SingleByIdAsync<T>(id);
    }

    public async Task<int> InsertAsync(T entity)
    {
        entity.CreatedAt = DateTime.UtcNow;
        using var db = _factory.CreateConnection();
        return (int)await db.InsertAsync(entity, selectIdentity: true);
    }

    public async Task<bool> UpdateAsync(T entity)
    {
        entity.UpdatedAt = DateTime.UtcNow;
        using var db = _factory.CreateConnection();
        return await db.UpdateAsync(entity) > 0;
    }
    // ...pozostałe metody
}

JwtService — generowanie tokenów

csharp
// Infrastructure/Auth/JwtService.cs
public class JwtService : IJwtService
{
    private readonly JwtSettings _settings;

    public JwtService(IOptions<JwtSettings> settings) => _settings = settings.Value;

    public string GenerateToken(User user)
    {
        var claims = new List<Claim>
        {
            new(ClaimTypes.NameIdentifier, user.Id.ToString()),
            new(ClaimTypes.Email, user.Email),
            new(ClaimTypes.Role, user.Role),
            new(ClaimTypes.GivenName, user.FirstName),
            new("fullName", $"{user.FirstName} {user.LastName}")
        };

        var identity = new ClaimsIdentity(claims, JwtBearerDefaults.AuthenticationScheme);
        var key   = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_settings.SecretKey));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer:            _settings.Issuer,
            audience:          _settings.Audience,
            claims:            identity.Claims,
            expires:           DateTime.UtcNow.AddHours(_settings.ExpiryHours),
            signingCredentials: creds
        );

        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}

{
    public DietProfile()
    {
        CreateMap<Diet, DietDto>()
            .ForMember(d => d.VariantCount, o => o.MapFrom(s => s.Variants.Count));

        CreateMap<CreateDietRequest, Diet>()

Caching

csharp
// Infrastructure/Cache/ICacheService.cs
public interface ICacheService
{
    Task<T?> GetAsync<T>(string key);
    Task     SetAsync<T>(string key, T value, TimeSpan? expiry = null);
    Task     RemoveAsync(string key);
    Task     RemoveByPrefixAsync(string prefix);
}

// Infrastructure/Cache/MemoryCacheService.cs
public class MemoryCacheService : ICacheService
{
    private readonly IMemoryCache _cache;
    private readonly HashSet<string> _keys = new();

    public MemoryCacheService(IMemoryCache cache) => _cache = cache;

    public Task<T?> GetAsync<T>(string key)
        => Task.FromResult(_cache.TryGetValue(key, out T? value) ? value : default);

    public Task SetAsync<T>(string key, T value, TimeSpan? expiry = null)
    {
        _keys.Add(key);
        var options = new MemoryCacheEntryOptions
            { AbsoluteExpirationRelativeToNow = expiry ?? TimeSpan.FromMinutes(10) };
        _cache.Set(key, value, options);
        return Task.CompletedTask;
    }

    public Task RemoveByPrefixAsync(string prefix)
    {
        foreach (var key in _keys.Where(k => k.StartsWith(prefix)))
            _cache.Remove(key);
        return Task.CompletedTask;
    }
}

Infrastructure/DependencyInjection.cs

csharp
public static class DependencyInjection
{
    public static IServiceCollection AddInfrastructure(
        this IServiceCollection services, IConfiguration config)
    {
        // Connection Factory
        services.AddSingleton<IDbConnectionFactory, SqliteConnectionFactory>();

        // Repositories
        services.AddScoped<IUserRepository,  UserRepository>();
        services.AddScoped<IDietRepository,  DietRepository>();
        services.AddScoped<IOrderRepository, OrderRepository>();

        // FluentMigrator
        services
            .AddFluentMigratorCore()
            .ConfigureRunner(rb => rb
                .AddSQLite()
                .WithGlobalConnectionString(config.GetConnectionString("DefaultConnection"))
                .ScanIn(typeof(DependencyInjection).Assembly).For.Migrations())
            .AddLogging(lb => lb.AddFluentMigratorConsole());

        // Auth / Cache / External services
        services.Configure<JwtSettings>(config.GetSection("JwtSettings"));
        services.AddScoped<IJwtService, JwtService>();
        services.AddMemoryCache();
        services.AddSingleton<ICacheService, MemoryCacheService>();

Obsługa Plików

// Infrastructure/FileStorage/LocalFileStorageService.cs
public class LocalFileStorageService : IFileStorageService
{
    private readonly string _uploadPath;

    public LocalFileStorageService(IWebHostEnvironment env)
        => _uploadPath = Path.Combine(env.WebRootPath, "uploads");

    public async Task<string> SaveAsync(IFormFile file, string subfolder)
    {
        var allowed = new[] { ".jpg", ".jpeg", ".png", ".webp" };
        var ext = Path.GetExtension(file.FileName).ToLowerInvariant();
        if (!allowed.Contains(ext))
            throw new InvalidOperationException("Nieobsługiwany format pliku.");

        var fileName = $"{Guid.NewGuid()}{ext}";
        var dir = Path.Combine(_uploadPath, subfolder);
        Directory.CreateDirectory(dir);

        var fullPath = Path.Combine(dir, fileName);
        await using var stream = File.Create(fullPath);
        await file.CopyToAsync(stream);

        return $"/uploads/{subfolder}/{fileName}";
    }
}

Ważne kwestie do omówienia z zespołem

1️⃣
Dapper vs EF Core — Dapper nie ma lazy loadingu. Przy ManyToMany trzeba jawnie ładować dane przez JOIN lub dodatkowe zapytania. Napiszcie te zapytania "na papierze" przed implementacją modułu produkcyjnego.
2️⃣
SQLite write-lock — SQLite blokuje całą bazę przy każdym zapisie. Dla projektu akademickiego wystarczy; upgrade do PostgreSQL wymaga tylko podmiany providera w DI.
3️⃣
FluentMigrator jako jedyne źródło DDL — atrybuty encji nie tworzą tabel. Migracje to JEDYNE miejsce DDL.
4️⃣
JWT w cookie vs localStorage — W MVC (Razor Views) tokeny wygodniej trzymać w HttpOnly cookie. [Authorize] działa automatycznie bez ręcznego dołączania headera.
5️⃣
QuestPDF licencja — Od wersji 2023.x darmowy dla projektów community. Wymagane: QuestPDF.Settings.License = LicenseType.Community;
6️⃣
Sekrety — NIE commitujcie kluczy API do Git. Używajcie dotnet user-secrets lokalnie i zmiennych środowiskowych na produkcji.
